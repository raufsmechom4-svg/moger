                   hl.FillTransparency = 0.5
                   hl.OutlineTransparency = 0

               hl.OutlineColor = c
           else
               if hl then
                   pcall(function() hl:Destroy() end)
                   ChamsHighlights[p] = nil
               end
           end
       end
   end

local ArmsScan = {Parts = {}, Textures = {}, Last = 0}

local function RefreshArmsScan()
   local now = tick()
   if now - ArmsScan.Last < 0.5 then return end
   ArmsScan.Last = now
   local parts, texs = {}, {}
   for _, obj in ipairs(Camera:GetDescendants()) do
       if obj:IsA("BasePart") then
           parts[#parts + 1] = obj
       elseif obj:IsA("Texture") or obj:IsA("Decal") or obj:IsA("SurfaceAppearance") then
           texs[#texs + 1] = obj
       end
   end
   ArmsScan.Parts = parts
   ArmsScan.Textures = texs
end

local function ApplyArmsNoTextures()
   local col = GetArmsNoTexColor()
   for _, obj in ipairs(ArmsScan.Textures) do
       if not ArmsTexOriginals[obj] then
           ArmsTexOriginals[obj] = obj.Parent
           pcall(function() obj.Parent = nil end)
       end
   end
   if not ArmsChams.Enabled then
       for _, obj in ipairs(ArmsScan.Parts) do
           if not ArmsOriginals[obj] then
               ArmsOriginals[obj] = {Color = obj.Color, Material = obj.Material, Transparency = obj.Transparency}
           end
           pcall(function()
               obj.Color = col
           end)
       end
   end
end

local function ClearArmsNoTextures()
   for obj, parent in pairs(ArmsTexOriginals) do
       pcall(function()
           if obj and obj.Parent == nil and parent then obj.Parent = parent end
       end)
   end
   ArmsTexOriginals = {}
   if not ArmsChams.Enabled then
       ClearArmsChams()
   end
end

local function IsGlovePart(part)
   local node = part.Parent
   while node and node ~= Camera do
       if string.lower(node.Name):find("glove") then return true end
       node = node.Parent
   end
   return string.lower(part.Name):find("glove") ~= nil
end

function ApplyArmsChams()
   local col = GetArmsColor()
   local mat = Enum.Material[ArmsChams.Material] or Enum.Material.ForceField
   for _, part in ipairs(ArmsScan.Parts) do
       if part:IsA("BasePart") then
           local isGlove = IsGlovePart(part)
           if ArmsChams.HideGloves and isGlove then
               if not ArmsOriginals[part] then
                   ArmsOriginals[part] = {Color = part.Color, Material = part.Material, Transparency = part.Transparency, LTM = part.LocalTransparencyModifier}
               end
               pcall(function() part.LocalTransparencyModifier = 1 end)
           else
               if not ArmsOriginals[part] then
                   ArmsOriginals[part] = {Color = part.Color, Material = part.Material, Transparency = part.Transparency, LTM = part.LocalTransparencyModifier}
               end
               pcall(function()
                   part.LocalTransparencyModifier = ArmsOriginals[part].LTM or 0
                   part.Color = col
                   part.Material = mat
                   part.Transparency = ArmsChams.Transparency
               end)
           end
       end
   end
end

ClearArmsChams = function()
   for part, props in pairs(ArmsOriginals) do
       pcall(function()
           if part and part.Parent then
               part.Color = props.Color
               part.Material = props.Material
               part.Transparency = props.Transparency
               if props.LTM ~= nil then part.LocalTransparencyModifier = props.LTM end
           end
       end)
   end
   ArmsOriginals = {}
end

function IsCharacterPart(part)
   if part:IsDescendantOf(Camera) then return true end
   local model = part:FindFirstAncestorOfClass("Model")
   while model do
       if model:FindFirstChildOfClass("Humanoid") or Players:GetPlayerFromCharacter(model) then
           return true
       end
       model = model.Parent and model.Parent:FindFirstAncestorOfClass("Model")
   end
   return false
end

function ApplyWorldChams()
   local col = GetWorldColor()
   for _, obj in ipairs(Workspace:GetDescendants()) do
       if obj:IsA("BasePart") and not obj:IsA("Terrain") then
           if not IsCharacterPart(obj) then
               if not WorldOriginals[obj] then
                   WorldOriginals[obj] = {Color = obj.Color, Material = obj.Material}
               end
               pcall(function()
                   obj.Color = col
                   obj.Material = Enum.Material.SmoothPlastic
               end)
           end
       elseif WorldChams.NoTextures and (obj:IsA("Texture") or obj:IsA("Decal") or obj:IsA("SurfaceAppearance")) then
           local parentPart = obj.Parent
           if parentPart and parentPart:IsA("BasePart") and not IsCharacterPart(parentPart) then
               if not WorldOriginals[obj] then
                   WorldOriginals[obj] = {Destroyed = true, Parent = parentPart}
                   pcall(function() obj.Parent = nil end)
               end
           end
       end
   end
end

function ClearWorldChams()
   for obj, props in pairs(WorldOriginals) do
       pcall(function()
           if props.Destroyed then
               if obj and obj.Parent == nil then obj.Parent = props.Parent end
           elseif obj and obj.Parent then
               obj.Color = props.Color
               obj.Material = props.Material
           end
       end)
   end
   WorldOriginals = {}
end

       if now - WorldLastScan > 2.5 then
           WorldLastScan = now
           ApplyWorldChams()
       end
   elseif next(WorldOriginals) then
       ClearWorldChams()
   end
end)

local function GetMuzzlePos()
   local char = LocalPlayer.Character
   if not char then return nil end
   local gun = char:FindFirstChild("Gun")
   if gun then
       local flash = gun:FindFirstChild("Flash")
       if flash then
           local ok, pos = pcall(function() return flash.WorldPosition end)
           if ok and pos then return pos end
       end
       if gun:IsA("BasePart") then return gun.Position end
       local handle = gun:FindFirstChild("Handle") or gun:FindFirstChildWhichIsA("BasePart", true)
       if handle then return handle.Position end
   end
   local tool = char:FindFirstChildOfClass("Tool")
   if tool then
       local handle = tool:FindFirstChild("Handle") or tool:FindFirstChildWhichIsA("BasePart", true)
       if handle then return handle.Position end
   end
   local head = char:FindFirstChild("Head")
   if head then return head.Position end
   return nil
end

local function CreateTracer(hitPos)
   if not Tracers.Enabled then return end
   local muzzle = GetMuzzlePos()
   if not muzzle then return end
   pcall(function()
       local color = GetTracerColor()
       local p0 = CreateTemporaryPart(muzzle)
       local p1 = CreateTemporaryPart(hitPos)
       local beam = Instance.new("Beam")
       beam.Texture = TracerTextures[Tracers.Texture] or TracerTextures.Line
       local a0 = Instance.new("Attachment")
       a0.Parent = p0
       local a1 = Instance.new("Attachment")
       a1.Parent = p1
       beam.Attachment0 = a0
       beam.Attachment1 = a1
       beam.Color = ColorSequence.new({
           ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
           ColorSequenceKeypoint.new(1, color)
       })
       beam.Width0 = 0.1 + Tracers.Thickness * 0.05
       beam.Width1 = 0.1 + Tracers.Thickness * 0.05
       beam.FaceCamera = true
       beam.LightEmission = 1
       beam.LightInfluence = 0
       beam.Transparency = NumberSequence.new(0)
       beam.Parent = GetDebrisParent()
       task.spawn(function()
           task.wait(Tracers.Duration)
           for i = 1, 10 do
               beam.Transparency = NumberSequence.new(i / 10)
               task.wait(0.05)
           end
           beam:Destroy()
           a0:Destroy()
           a1:Destroy()
           p0:Destroy()
           p1:Destroy()
       end)
   end)
end

           p:Destroy()
       end)
   end)
end

   if now - lastTracerSpawn < 0.05 then return end
   lastTracerSpawn = now
   CreateTracer(hitPos)
   CreateImpact(hitPos)
end

local function getShucky()
   if Shucky then return Shucky end
   local ok, mod = pcall(function()
       local shuckyModule = ReplicatedStorage:WaitForChild("Modules"):WaitForChild("Shucky")
       return require(shuckyModule)
   end)
   if ok then
       Shucky = mod
       if not ShuckyHooked and hookfunction and typeof(Shucky.activateresult) == "function" then
           local hookOk = pcall(function()
               local oldActivate
               local activateHook = function(hitData, weapon, originPos, ...)
                   if Tracers.Enabled and not FireEventHooked and hitData and hitData[1] then
                       pcall(function()
                           SpawnTracer(hitData[1].Position)
                       end)
                   end
                   return oldActivate(hitData, weapon, originPos, ...)
               end
               if newcclosure then activateHook = newcclosure(activateHook) end
               oldActivate = hookfunction(Shucky.activateresult, activateHook)
           end)
           if hookOk then
               ShuckyHooked = true
               warn("[Tracers] Shucky hook installed")
           else
               warn("[Tracers] Shucky hook failed, using fallback")
           end
       end
   end
   return Shucky
end

local function ResolveFireEvent()
   if FireEvent then return FireEvent end
   pcall(function()
       local shucky = getShucky()
       if shucky and shucky.activateresult and debug and debug.getupvalue and setthreadidentity and getthreadidentity then
           local id = getthreadidentity()
           setthreadidentity(2)
           local ok, fn = pcall(debug.getupvalue, shucky.activateresult, 5)
           if ok and type(fn) == "function" then
               local ok2, remote = pcall(fn)
               if ok2 and typeof(remote) == "Instance" then
                   FireEvent = remote
               end
           end
           setthreadidentity(id)
       end
   end)
   if not FireEvent then
       pcall(function()
           local seed = ReplicatedStorage:GetAttribute("AmongUsSauce")
           if type(seed) == "number" then
               local t, chars = math.max(1, math.floor(seed) % 2147483647), {}
               for i = 1, 24 do
                   t = t * 48271 % 2147483647
                   local w = t % 62 + 1
                   chars[i] = string.sub("abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789", w, w)
               end
               local remoteName = "sus_" .. table.concat(chars)
               local events = ReplicatedStorage:FindFirstChild("Events")
               if events then
                   local r = events:FindFirstChild(remoteName)
                   if r then FireEvent = r end
               end
           end
       end)
   end
   return FireEvent
end

local function defusalFire(targetPos, targetPart, burstCount, wallbang)
   local char = LocalPlayer.Character
   if not char then return end
   local head = char:FindFirstChild("Head")
   if not head then return end
   local gunName = char:GetAttribute("WhatGun")
   if not gunName then return end
   local shucky = getShucky()
   if not shucky then return end
   local okWeapon, weapon = pcall(function()
       return shucky.GetWep(gunName, LocalPlayer)
   end)
   if not okWeapon or not weapon then return end
   local burst = math.max(1, burstCount or 1)
   local wb = wallbang and true or nil
   local hitData = {}
   for i = 1, burst do
       hitData[i] = {Hit = targetPart, Position = targetPos, Normal = Vector3.new(0, 1, 0), Wallbang = wb}
   end
   SpawnTracer(targetPos)
   if targetPart then
       pcall(function()
           local model = targetPart:FindFirstAncestorOfClass("Model")
           local plr = model and Players:GetPlayerFromCharacter(model)
           if plr then RecentHits[plr] = tick() end
       end)
   end
   local spoof = SpoofTargetTeam(targetPart)
   if spoof then
       local remote = ResolveFireEvent()
       local method = SilentAim.FFMethod
       local function fireDirect(data)
           if remote and remote.FireServer then
               pcall(function()
                   remote:FireServer(data, weapon, head.Position, nil, nil, nil, nil)
               end)
           end
       end
       local function fireShucky(data)
           pcall(function()
               shucky.activateresult(data, weapon, head.Position, nil, nil, nil, nil)
           end)
       end
       if method == "Direct" then
           fireDirect(hitData)
       elseif method == "Shucky" then
           fireShucky(hitData)
       elseif method == "Both" then
           fireDirect(hitData)
           fireShucky(hitData)
       elseif method == "Wallbang" then
           local wbData = {{Hit = targetPart, Position = targetPos, Normal = Vector3.new(0, 1, 0), Wallbang = true}}
           fireDirect(wbData)
           fireShucky(wbData)
       elseif method == "Multi" then
           local model = targetPart and targetPart:FindFirstAncestorOfClass("Model")
           local parts = {}
           if model then
               for _, pn in ipairs({"Head", "HumanoidRootPart", "UpperTorso", "Torso", "Left Arm", "Right Arm", "Left Leg", "Right Leg", "LeftUpperArm", "RightUpperArm", "LeftUpperLeg", "RightUpperLeg"}) do
                   local part = model:FindFirstChild(pn)
                   if part and part:IsA("BasePart") then
                       table.insert(parts, {Hit = part, Position = part.Position, Normal = Vector3.new(0, 1, 0), Wallbang = nil})
                   end
               end
           end
           if #parts == 0 then parts = hitData end
           fireDirect(parts)
           fireShucky(parts)
       end
   else
       pcall(function()
           shucky.activateresult(hitData, weapon, head.Position, nil, nil, nil, nil)
       end)
   end
   RestoreTargetTeam(spoof)
end

IsTeammate = function(player)
   if player == LocalPlayer then return true end
   if ReplicatedStorage:GetAttribute("NoTKPenalty") then return false end
   local pt1 = LocalPlayer:GetAttribute("Team")
   local pt2 = player:GetAttribute("Team")
   if pt1 and pt2 then return tostring(pt1) == tostring(pt2) end
   if player.Team and LocalPlayer.Team and player.Team == LocalPlayer.Team then
       return true
   end
   local c1 = LocalPlayer.Character
   local c2 = player.Character
   if c1 and c2 then
       local a1 = c1:GetAttribute("Team")
       local a2 = c2:GetAttribute("Team")
       if a1 and a2 and tostring(a1) == tostring(a2) then return true end
       local tv1 = c1:FindFirstChild("TeamValue") or c1:FindFirstChild("Team")
       local tv2 = c2:FindFirstChild("TeamValue") or c2:FindFirstChild("Team")
       if tv1 and tv2 and tv1:IsA("ValueBase") and tv2:IsA("ValueBase") and tv1.Value == tv2.Value then
           return true
       end
   end
   local pt1 = LocalPlayer:GetAttribute("Team")
   local pt2 = player:GetAttribute("Team")
   if pt1 and pt2 and tostring(pt1) == tostring(pt2) then return true end
   return false
end

local BombText = NewDrawing("Text", {Size = 15, Color = Color3.fromRGB(255,60,60), Center = true, Outline = true, OutlineColor = Color3.fromRGB(0,0,0), Transparency = 1, Visible = false})

local BombCache = {Part = nil, LastScan = 0}

local function FindBomb()
   local now = tick()
   if now - BombCache.LastScan < 5 then
       if BombCache.Part and BombCache.Part.Parent then return BombCache.Part end
   end
   BombCache.LastScan = now
   BombCache.Part = nil
   local names = {bomb = true, c4 = true, plantedbomb = true, bombpl = true}
   for _, obj in ipairs(Workspace:GetDescendants()) do
       local n = string.lower(obj.Name)
       if names[n] then
           if obj:IsA("BasePart") then
               BombCache.Part = obj
               break
           elseif obj:IsA("Model") then
               local pp = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)
               if pp then
                   BombCache.Part = pp
                   break
               end
           end
       end
   end
   return BombCache.Part
end

   end
   local part = FindBomb()

       BombText.Visible = false
       return
   end
   local pos, onScreen = Camera:WorldToViewportPoint(part.Position)
   if not onScreen then
       BombText.Visible = false
       return
   end
   local dist = (part.Position - Camera.CFrame.Position).Magnitude
   local label = "BOMB [" .. tostring(math.floor(dist)) .. "m]"
   local timer = nil
   pcall(function()
       local parent = part.Parent
       timer = part:GetAttribute("Timer") or part:GetAttribute("BombTimer") or part:GetAttribute("ExplodeTime")
       if not timer and parent then
           timer = parent:GetAttribute("Timer") or parent:GetAttribute("BombTimer") or parent:GetAttribute("ExplodeTime")
       end
   end)
   local timerNum = timer and tonumber(timer) or nil
   if timerNum then
       label = "BOMB [" .. tostring(math.floor(dist)) .. "m] " .. string.format("%.1f", timerNum) .. "s"
   end
   BombText.Text = label
   BombText.Position = Vector2.new(pos.X, pos.Y)
   BombText.Visible = true
end

local function IsVisible(targetPart)
   local origin = Camera.CFrame.Position
   local direction = targetPart.Position - origin
   local rayParams = RaycastParams.new()
   rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
   rayParams.FilterType = Enum.RaycastFilterType.Exclude
   local result = Workspace:Raycast(origin, direction, rayParams)
   if not result then return true end
   local hitModel = result.Instance:FindFirstAncestorOfClass("Model")
   local targetModel = targetPart:FindFirstAncestorOfClass("Model")
   return hitModel == targetModel
end

local BodyPartsBuf = {}

local function GetSilentAimTarget()
   local vp = Camera.ViewportSize
   local centerX = vp.X / 2
   local centerY = vp.Y / 2
   local fovR = SilentAim.FOV
   local fovR2 = fovR * fovR
   local checkVis = SilentAim.VisibilityCheck
   local hitPart = SilentAim.HitPart
   local predict = SilentAim.Prediction
   local pFac = SilentAim.PredictionFactor
   local multi = SilentAim.MultiPoint
   local tMode = SilentAim.TargetMode
   local camPos = Camera.CFrame.Position
   local bestPart, bestPos, bestScore = nil, nil, math.huge

   for _, p in ipairs(Players:GetPlayers()) do
       if p ~= LocalPlayer then
           if SilentAim.TeamCheck and not SilentAim.FriendlyFire and IsTeammate(p) then continue end
           local data = GetCharDataFast(p)
           if data and IsPlayerAlive(p) then
               local parts
               if hitPart == "Random" then
                   parts = {data.Head, data.HRP}
               elseif hitPart == "Closest" then
                   parts = {data.Head, data.HRP, data.Character:FindFirstChild("UpperTorso"), data.Character:FindFirstChild("Torso")}
               elseif hitPart == "Body" then
                   local n = 0
                   for _, part in ipairs(data.Character:GetChildren()) do
                       if part:IsA("BasePart") then
                           n = n + 1
                           BodyPartsBuf[n] = part
                       end
                   end
                   for i = n + 1, #BodyPartsBuf do BodyPartsBuf[i] = nil end
                   parts = BodyPartsBuf
               else
                   parts = {data.Character:FindFirstChild(hitPart) or data.Head}
               end

               local cand = {}
               local ncand = 0
               for _, part in ipairs(parts) do
                   if part then
                       local aimPos = part.Position
                       if multi then
                           local s = part.Size
                           aimPos = aimPos + Vector3.new(
                               (math.random() - 0.5) * s.X * 0.4,
                               (math.random() - 0.5) * s.Y * 0.4,
                               (math.random() - 0.5) * s.Z * 0.4
                           )
                       end
                       if predict then
                           aimPos = aimPos + part.AssemblyLinearVelocity * pFac
                       end
                       local sp, onScreen = Camera:WorldToViewportPoint(aimPos)
                       if onScreen then
                           local dx = sp.X - centerX
                           local dy = sp.Y - centerY
                           local d2 = dx * dx + dy * dy
                           if d2 <= fovR2 then
                               local score = d2
                               if tMode == "Distance" then
                                   score = (part.Position - camPos).Magnitude
                               elseif tMode == "Health" then
                                   score = GetHealth(p, data.Humanoid)
                               end
                               ncand = ncand + 1
                               cand[ncand] = {part, aimPos, score}
                           end
                       end
                   end
               end

               if ncand > 0 then
                   if ncand > 1 then
                       table.sort(cand, function(a, b) return a[3] < b[3] end)
                   end
                   if checkVis then
                       for i = 1, ncand do
                           local c0 = cand[i]
                           if IsVisible(c0[1]) then
                               if c0[3] < bestScore then
                                   bestPart, bestPos, bestScore = c0[1], c0[2], c0[3]
                               end
                               break
                           end
                       end
                   else
                       local c0 = cand[1]
                       if c0[3] < bestScore then
                           bestPart, bestPos, bestScore = c0[1], c0[2], c0[3]
                       end
                   end
               end
           end
       end
   end
   return bestPart, bestPos
end

local function ShouldSilentAim()
   if not SilentAim.Enabled then return false end
   if SilentAim.Bind == "Always" then return true end
   if SilentAim.Bind == "LMB" then
       return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
   end
   if SilentAim.Bind == "RMB" then
       return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
   end
   return false
end

local function CheckTargetState(part)
   local origin = Camera.CFrame.Position
   local dir = part.Position - origin
   local dist = dir.Magnitude
   local rp = RaycastParams.new()
   rp.FilterDescendantsInstances = {LocalPlayer.Character}
   rp.FilterType = Enum.RaycastFilterType.Exclude
   local hit = Workspace:Raycast(origin, dir, rp)
   if not hit then return "visible" end
   local hitModel = hit.Instance:FindFirstAncestorOfClass("Model")
   local targetModel = part:FindFirstAncestorOfClass("Model")
   if hitModel == targetModel then return "visible" end
   if dist > SilentAim.WBMaxDist then return "none" end
   local rp2 = RaycastParams.new()
   rp2.FilterDescendantsInstances = {LocalPlayer.Character, hit.Instance}
   rp2.FilterType = Enum.RaycastFilterType.Exclude
   local depth = dist + 10
   local hit2 = Workspace:Raycast(origin + dir.Unit * depth, -dir.Unit * depth, rp2)
   local thickness = math.huge
   if hit2 and hit2.Instance == hit.Instance then
       thickness = (hit2.Position - hit.Position).Magnitude
   end
   if thickness <= SilentAim.WBMaxThickness then return "wallbang" end
   return "none"
end

local function GetFireDelay()
   local d = SilentAim.AutoFireDelay
   if SilentAim.Humanize then
       d = d * (0.75 + math.random() * 0.5)
   end
   return d
end

local function PickAimPart(p, data)
   local parts
   local hitPart = SilentAim.HitPart
   if hitPart == "Random" then
       parts = {data.Head, data.HRP}
   elseif hitPart == "Closest" then
       parts = {data.Head, data.HRP, data.Character:FindFirstChild("UpperTorso"), data.Character:FindFirstChild("Torso")}
   elseif hitPart == "Body" then
       local n = 0
       for _, part in ipairs(data.Character:GetChildren()) do
           if part:IsA("BasePart") then
               n = n + 1
               BodyPartsBuf[n] = part
           end
       end
       for i = n + 1, #BodyPartsBuf do BodyPartsBuf[i] = nil end
       parts = BodyPartsBuf
   else
       parts = {data.Character:FindFirstChild(hitPart) or data.Head}
   end
   local vp = Camera.ViewportSize
   local cx = vp.X / 2
   local cy = vp.Y / 2
   local fovR2 = SilentAim.FOV * SilentAim.FOV
   local predict = SilentAim.Prediction
   local pFac = SilentAim.PredictionFactor
   local bestPart, bestD2 = nil, math.huge
   for _, part in ipairs(parts) do
       if part then
           local aimPos = part.Position
           if predict then aimPos = aimPos + part.AssemblyLinearVelocity * pFac end
           local sp, onScreen = Camera:WorldToViewportPoint(aimPos)
           if onScreen then
               local dx = sp.X - cx
               local dy = sp.Y - cy
               local d2 = dx * dx + dy * dy
               if d2 <= fovR2 and d2 < bestD2 then
                   bestPart, bestD2 = part, d2
               end
           end
       end
   end
   return bestPart
end

local FireQueue = {}

local function BuildFireQueue()
   FireQueue = {}
   local maxT = math.max(1, SilentAim.MaxTargets)
   local added = 0
   for _, p in ipairs(Players:GetPlayers()) do
       if added >= maxT then break end
       if p ~= LocalPlayer then
           if SilentAim.TeamCheck and not SilentAim.FriendlyFire and IsTeammate(p) then continue end
           local data = GetCharDataFast(p)
           if data and IsPlayerAlive(p) then
               local part = PickAimPart(p, data)
               if part then
                   local state = "visible"
                   if SilentAim.VisibilityCheck then state = CheckTargetState(part) end
                   if state == "visible" then
                       FireQueue[#FireQueue + 1] = {part, false}
                       added = added + 1
                   elseif state == "wallbang" and SilentAim.SmartWallbang then
                       FireQueue[#FireQueue + 1] = {part, true}
                       added = added + 1
                   end
               end
           end
       end
   end
end

local function GetNearestEnemy()
   local closest, minDist = nil, math.huge
   for _, p in ipairs(Players:GetPlayers()) do
       if p ~= LocalPlayer and not IsTeammate(p) then
           local data = GetCharDataFast(p)
           if data and IsPlayerAlive(p) then
               local dist = (data.HRP.Position - Camera.CFrame.Position).Magnitude
               if dist < minDist then
                   minDist = dist
                   closest = data.HRP
               end
           end
       end
   end
   return closest
end

local function GetCharacterJoints(char)
   local joints = {}
   if not char then return joints end
   local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
   if not torso then return joints end

   local neck = torso:FindFirstChild("Neck")
   if not neck then
       local head = char:FindFirstChild("Head")
       if head then neck = head:FindFirstChild("Neck") end
   end
   if neck and neck:IsA("Motor6D") then joints.Neck = neck end

   local waist = torso:FindFirstChild("Waist")
   if waist and waist:IsA("Motor6D") then joints.Waist = waist end

   local hrp = char:FindFirstChild("HumanoidRootPart")
   local root = nil
   if hrp then
       root = hrp:FindFirstChild("RootJoint") or hrp:FindFirstChild("Root") or hrp:FindFirstChild("Root Joint")
   end
   if not root then
       local lower = char:FindFirstChild("LowerTorso")
       if lower then
           root = lower:FindFirstChild("Root") or lower:FindFirstChild("RootJoint")
       end
   end
   if not root then
       root = torso:FindFirstChild("RootJoint") or torso:FindFirstChild("Root") or torso:FindFirstChild("Root Joint")
   end
   if root and root:IsA("Motor6D") then joints.Root = root end

   local torso6 = char:FindFirstChild("Torso")
   if torso6 then
       local lh = torso6:FindFirstChild("Left Hip")
       local rh = torso6:FindFirstChild("Right Hip")
       if lh and lh:IsA("Motor6D") then joints.LeftHip = lh end
       if rh and rh:IsA("Motor6D") then joints.RightHip = rh end
   end
   return joints
end

local function RotJoint(joint, origC0, rotCF, pivotOffset)
   local piv = origC0.Position + (pivotOffset or Vector3.zero)
   joint.C0 = CFrame.new(piv) * rotCF * CFrame.new(-piv) * origC0
end

local function ApplyPitchToJoints(char, pitchRad, rollRad)
   local joints = GetCharacterJoints(char)
   local half = pitchRad * 0.5
   local halfRoll = rollRad * 0.5
   if joints.Waist and OriginalC0.Waist then
       RotJoint(joints.Waist, OriginalC0.Waist, CFrame.Angles(half, 0, halfRoll))
   elseif joints.Root and OriginalC0.Root then
       RotJoint(joints.Root, OriginalC0.Root, CFrame.Angles(half, 0, halfRoll), Vector3.new(0, -1, 0))
       if joints.LeftHip and OriginalC0.LeftHip then
           RotJoint(joints.LeftHip, OriginalC0.LeftHip, CFrame.Angles(-half, 0, -halfRoll))
       end
       if joints.RightHip and OriginalC0.RightHip then
           RotJoint(joints.RightHip, OriginalC0.RightHip, CFrame.Angles(-half, 0, -halfRoll))
       end
   end
   if joints.Neck and OriginalC0.Neck then
       RotJoint(joints.Neck, OriginalC0.Neck, CFrame.Angles(pitchRad * 0.35, 0, halfRoll))
   end
end

local function ResetJoints(char)
   local joints = GetCharacterJoints(char)
   if joints.Neck and OriginalC0.Neck then joints.Neck.C0 = OriginalC0.Neck end
   if joints.Waist and OriginalC0.Waist then joints.Waist.C0 = OriginalC0.Waist end
   if joints.Root and OriginalC0.Root then joints.Root.C0 = OriginalC0.Root end
   if joints.LeftHip and OriginalC0.LeftHip then joints.LeftHip.C0 = OriginalC0.LeftHip end
   if joints.RightHip and OriginalC0.RightHip then joints.RightHip.C0 = OriginalC0.RightHip end
end

local AA_State = {
   Yaw = nil,
   Roll = 0,
   Pitch = 0,
   SpinAngle = 0,
   LastJitterTime = 0,
   YawJitterOffset = 0,
   PitchJitterOffset = 0
}

local function AA_ComputeYaw(dt)
   local camY = -math.atan2(Camera.CFrame.LookVector.Z, Camera.CFrame.LookVector.X) + math.rad(-90)
   local mode = AntiAim.YawMode

   if mode == "AtTargets" then
       local nearest = GetNearestEnemy()
       if nearest then
           local dir = (nearest.Position - Cache.HRP.Position).Unit
           return math.atan2(dir.X, dir.Z) + math.rad(180)
       end
       return camY + math.rad(180)
   elseif mode == "Backwards" then
       return camY + math.rad(180)
   elseif mode == "Left" then
       return camY + math.rad(90)
   elseif mode == "Right" then
       return camY - math.rad(90)
   elseif mode == "Custom" then
       return camY + math.rad(AntiAim.YawAngle)
   elseif mode == "Random" then
       return camY + math.rad(math.random(0, 360))
   elseif mode == "Spin" then
       AA_State.SpinAngle = (AA_State.SpinAngle + AntiAim.SpinSpeed * dt * 60) % 360
       return camY + math.rad(AA_State.SpinAngle)
   end
   return camY
end

local function AA_ComputePitch()
   local mode = AntiAim.PitchMode
   if mode == "Up" then return math.rad(89) end
   if mode == "Down" then return math.rad(-89) end
   if mode == "Zero" then return 0 end
   if mode == "Custom" then return math.rad(AntiAim.CustomPitch) end
   if mode == "Random" then return math.rad(math.random(-89, 89)) end
   return 0
end

local function AA_ComputeRoll()
   local mode = AntiAim.LeanMode
   if mode == "Left" then return math.rad(AntiAim.LeanAmount) end
   if mode == "Right" then return math.rad(-AntiAim.LeanAmount) end
   if mode == "Custom" then return math.rad(AntiAim.LeanAmount) end
   if mode == "Jitter" then return math.rad(math.random(-AntiAim.LeanAmount, AntiAim.LeanAmount)) end
   return 0
end

local function AA_UpdateJitters()
   if tick() - AA_State.LastJitterTime <= AntiAim.JitterDelay then return end
   if AntiAim.YawJitterMode == "Center" then
       AA_State.YawJitterOffset = math.rad(math.random(-AntiAim.YawJitterRange, AntiAim.YawJitterRange))
   elseif AntiAim.YawJitterMode == "Offset" then
       AA_State.YawJitterOffset = math.rad(math.random(0, AntiAim.YawJitterRange))
   else
       AA_State.YawJitterOffset = 0
   end
   if AntiAim.PitchJitterMode == "Center" then
       AA_State.PitchJitterOffset = math.rad(math.random(-25, 25))
   elseif AntiAim.PitchJitterMode == "Offset" then
       AA_State.PitchJitterOffset = math.rad(math.random(0, 50))
   else
       AA_State.PitchJitterOffset = 0
   end
   AA_State.LastJitterTime = tick()
end

   pcall(function()
       local align = Cache.HRP:FindFirstChildOfClass("AlignOrientation")

local AA_PitchDebugPrinted = false
local AA_LastPitchSend = 0

local function AA_SendPitch(pitchRad)
   if tick() - AA_LastPitchSend < 1/60 then return end
   AA_LastPitchSend = tick()
   local sentVia = nil
   if AA_Network and AA_Network.FireServer then
       local ok, err = pcall(function()
           AA_Network.FireServer(AA_RemoteName, pitchRad, true)
       end)
       if ok then
           sentVia = "module"
       elseif not AA_PitchDebugPrinted then
           warn("[AntiAim] Module FireServer error: " .. tostring(err))
       end
   end
   if AA_PitchRemoteDirect and AA_PitchRemoteDirect.Parent then
       local ok2 = pcall(function()
           AA_PitchRemoteDirect:FireServer(pitchRad, true)
       end)
       if ok2 then sentVia = (sentVia or "") .. "+direct" end
   end
   if not AA_PitchDebugPrinted then
       AA_PitchDebugPrinted = true
       if sentVia then
           warn("[AntiAim] Pitch sent via " .. sentVia .. ", value=" .. tostring(pitchRad))
       else
           warn("[AntiAim] Pitch NOT sent, no working channel")
       end
   end
end

GunModsOriginals = {}
GunModSpreadAttrs = {"Spread","CrouchSpread","FireSpread","LadderSpread","LandSpread","MoveSpread","StandSpread","JumpSpread","RunSpread"}
GunModAllAttrs = {"RecoilX","RecoilY","Spread","CrouchSpread","FireSpread","LadderSpread","LandSpread","MoveSpread","StandSpread","JumpSpread","RunSpread","Auto","FireRate","ScopedFireRate","UseAmmo"}

function GunModsTick()
   local weapons = ReplicatedStorage:FindFirstChild("Weapons")
   if not weapons then return end
   for _, folder in ipairs(weapons:GetChildren()) do
       if folder:IsA("Folder") then
           local orig = GunModsOriginals[folder]
           if not orig then
               orig = {}
               for _, a in ipairs(GunModAllAttrs) do
                   local v = folder:GetAttribute(a)
                   if v ~= nil then orig[a] = v end
               end
               GunModsOriginals[folder] = orig
           end
           local function setA(name, value)
               if folder:GetAttribute(name) ~= value then folder:SetAttribute(name, value) end
           end
           if GunMods.NoRecoil then
               setA("RecoilX", 0)
               setA("RecoilY", 0)
           else
               if orig.RecoilX ~= nil then setA("RecoilX", orig.RecoilX) end
               if orig.RecoilY ~= nil then setA("RecoilY", orig.RecoilY) end
           end
           if GunMods.NoSpread then
               for _, a in ipairs(GunModSpreadAttrs) do setA(a, 0) end
           else
               for _, a in ipairs(GunModSpreadAttrs) do
                   if orig[a] ~= nil then setA(a, orig[a]) end
               end
           end
           if GunMods.FullAuto then
               setA("Auto", true)
           else
               if orig.Auto ~= nil then setA("Auto", orig.Auto) end
           end
           if GunMods.FastFire then
               setA("FireRate", 0.01)
               setA("ScopedFireRate", 0.01)
           else
               if orig.FireRate ~= nil then setA("FireRate", orig.FireRate) end
               if orig.ScopedFireRate ~= nil then setA("ScopedFireRate", orig.ScopedFireRate) end
           end
           if GunMods.InfAmmo then
               setA("UseAmmo", 0)
           else
               if orig.UseAmmo ~= nil then setA("UseAmmo", orig.UseAmmo) end
           end
       end
   end
end

local lastCamLook = nil

RunService.Heartbeat:Connect(function(dt)
   local char = LocalPlayer.Character
   if not char then return end
   local hum = char:FindFirstChildOfClass("Humanoid")
   local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
   if not hum or not hrp then return end

   local onGround = hum.FloorMaterial ~= Enum.Material.Air
   local velocity = hrp.AssemblyLinearVelocity
   local moveVec = UserInputService:GetMoveVector()
   local moving = moveVec.Magnitude > 0.1

   if Movement.BhopEnabled then
       local wantJump = bhopHoldingSpace
       if Movement.BhopOnW and moving then
           wantJump = true
       end
       if wantJump and onGround then
           hum:ChangeState(Enum.HumanoidStateType.Jumping)
           hum.Jump = true
       end
   end

   if Movement.AutoStrafeEnabled and not onGround then
       local camCF = Camera.CFrame
       local look = Vector3.new(camCF.LookVector.X, 0, camCF.LookVector.Z)
       if look.Magnitude < 0.001 then look = Vector3.new(0, 0, -1) else look = look.Unit end
       local right = Vector3.new(camCF.RightVector.X, 0, camCF.RightVector.Z)
       if right.Magnitude < 0.001 then right = Vector3.new(1, 0, 0) else right = right.Unit end

       local input = Vector3.zero
       if moveVec.Z < -0.1 then input = input + look end
       if moveVec.Z > 0.1 then input = input - look end
       if moveVec.X < -0.1 then input = input - right end
       if moveVec.X > 0.1 then input = input + right end

       if input.Magnitude > 0.1 then
           local flatVel = Vector3.new(velocity.X, 0, velocity.Z)
           local camLookNow = Vector3.new(camCF.LookVector.X, 0, camCF.LookVector.Z)
           local yawDelta = 0
           if lastCamLook and camLookNow.Magnitude > 0.01 and lastCamLook.Magnitude > 0.01 then
               local a = lastCamLook.Unit
               local b = camLookNow.Unit
               yawDelta = -math.atan2(a.X * b.Z - a.Z * b.X, a.X * b.X + a.Z * b.Z)
           end
           lastCamLook = camLookNow

           local rotatedVel = CFrame.Angles(0, yawDelta, 0):VectorToWorldSpace(flatVel)
           local wishDir = input.Unit
           local targetSpeed = math.max(rotatedVel.Magnitude, 24 * (Movement.StrafeSpeed / 150))

           if Movement.ForwardBoost and moveVec.Z < -0.1 then
               targetSpeed = targetSpeed + Movement.StrafeSpeed * dt * 3
           end

           local newVel = wishDir * targetSpeed
           local finalVel = Vector3.new(newVel.X, velocity.Y, newVel.Z)

           local horiz = Vector3.new(finalVel.X, 0, finalVel.Z)
           if horiz.Magnitude > Movement.MaxAirSpeed then
               horiz = horiz.Unit * Movement.MaxAirSpeed
               finalVel = Vector3.new(horiz.X, finalVel.Y, horiz.Z)
           end

           hrp.AssemblyLinearVelocity = finalVel
       end
   else
       lastCamLook = Vector3.new(Camera.CFrame.LookVector.X, 0, Camera.CFrame.LookVector.Z)
   end
end)

local function DoRejoin()
   if rejoining then return end
   rejoining = true
   if AntiKick.Notify then warn("[AntiKick] Server kick detected, rejoining...") end
   task.wait(1)
   for attempt = 1, 20 do
       local ok, err
       if AntiKick.SameServer and SavedJobId and SavedJobId ~= "" then
           ok, err = pcall(function()
               TeleportService:TeleportToPlaceInstance(game.PlaceId, SavedJobId, LocalPlayer)
           end)
       else
           ok, err = pcall(function()
               TeleportService:Teleport(game.PlaceId, LocalPlayer)
           end)
       end
       if ok then
           if AntiKick.Notify then warn("[AntiKick] Rejoin teleport fired (attempt " .. attempt .. ")") end
           break
       end
       if AntiKick.Notify then warn("[AntiKick] Rejoin attempt " .. attempt .. " failed: " .. tostring(err)) end
       task.wait(2)
   end
   task.delay(30, function() rejoining = false end)
end

SkinChanger = {
   Enabled = false,
   Skin = "",
   CustomId = 0,
   _last = 0
}

SkinList = {}

function ScanSkins()
   SkinList = {}
   local seen = {}
   pcall(function()
       for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
           local id = nil
           local nm = obj.Name
           if obj:IsA("Texture") or obj:IsA("Decal") then
               id = obj.Texture
           elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
               id = obj.Image
           elseif obj:IsA("SpecialMesh") then
               id = obj.TextureId
           elseif obj:IsA("StringValue") and (obj.Value:match("rbxassetid://") or obj.Value:match("^%d+$")) then
               local v = obj.Value
               id = v:match("rbxassetid://") and v or ("rbxassetid://" .. v)
           end
           if id and id ~= "" and id:match("rbxassetid://%d+") then
               local parent = obj.Parent
               local pathName = ""
               local node = parent
               while node and node ~= ReplicatedStorage do
                   pathName = (node.Name .. "/" .. pathName)
                   node = node.Parent
               end
               if pathName:lower():find("skin") or nm:lower():find("skin") then
                   local label = nm
                   if label == "Texture" or label == "Image" or label == "Decal" then
                       label = parent and parent.Name or nm
                   end
                   if not seen[id] then
                       seen[id] = true
                       table.insert(SkinList, {Name = label, Id = id})
                   end
               end
           end
       end
   end)
   table.sort(SkinList, function(a, b) return a.Name < b.Name end)
   return SkinList
end

SkinOriginals = {}
SkinApplied = {}

function GetSkinId()
   if SkinChanger.Skin == "Custom" then
       local n = tonumber(SkinChanger.CustomId)
       if n and n > 0 then return "rbxassetid://" .. tostring(math.floor(n)) end
       return nil
   end
   for _, s in ipairs(SkinList) do
       if s.Name == SkinChanger.Skin then return s.Id end
   end
   return nil
end

function ClearSkinChanger()
   for obj, props in pairs(SkinOriginals) do
       pcall(function()
           if props.SpecialMesh and obj and obj.Parent then
               obj.TextureId = props.TextureId
           elseif obj and obj.Parent then
               obj:Destroy()
           end
       end)
   end
   SkinOriginals = {}
   SkinApplied = {}
end

function ApplySkinChanger()
   local id = GetSkinId()
   if not id then return end
   RefreshArmsScan()
   for _, part in ipairs(ArmsScan.Parts) do
       if part:IsA("BasePart") then
           local mesh = part:FindFirstChildOfClass("SpecialMesh")
           if mesh then
               if not SkinOriginals[mesh] then
                   SkinOriginals[mesh] = {SpecialMesh = true, TextureId = mesh.TextureId}
               end
               pcall(function() mesh.TextureId = id end)
               SkinApplied[part] = true
           else
               local hasTex = false
               for _, child in ipairs(part:GetChildren()) do
                   if child:IsA("Texture") and child:GetAttribute("SkinChangerTex") then
                       child.Texture = id
                       hasTex = true
                   end
               end
               if not hasTex and not SkinApplied[part] then
                   local tex = Instance.new("Texture")
                   tex.Texture = id
                   tex.StudsPerTileU = 1
                   tex.StudsPerTileV = 1
                   tex:SetAttribute("SkinChangerTex", true)
                   tex.Parent = part
                   SkinOriginals[tex] = {SpecialMesh = false}
                   SkinApplied[part] = true
               end
           end
       end
   end
end

   end

   for _, p in ipairs(JumpCircleParts) do pcall(function() p:Destroy() end) end
   JumpCircleParts = {}
   for i = 1, need do
       local p = Instance.new("Part")
       p.Name = "JumpRingFX"
       p.Anchored = true
       p.CanCollide = false
       p.CanQuery = false
       p.CanTouch = false
       p.Material = Enum.Material.Neon
       p.CastShadow = false
       p.Transparency = 1
       p.Size = Vector3.new(0.5, 0.08, 0.18)
       p.Parent = Workspace
       table.insert(JumpCircleParts, p)
   end
end

RunService.RenderStepped:Connect(function()
   if not JumpCircleEff.Enabled then
       if #JumpCircleParts > 0 then
           JumpCircleAlpha = math.max(0, JumpCircleAlpha - 0.08)
           for _, p in ipairs(JumpCircleParts) do pcall(function() p.Transparency = 1 - JumpCircleAlpha end) end
           if JumpCircleAlpha <= 0 then
               for _, p in ipairs(JumpCircleParts) do pcall(function() p:Destroy() end) end
               JumpCircleParts = {}
           end
       end
       return
   end
   pcall(function()
       EnsureJumpCircle()
       local char = LocalPlayer.Character
       local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
       local hum = char and char:FindFirstChildOfClass("Humanoid")
       if not hrp or not hum then return end
       local airborne = hum.FloorMaterial == Enum.Material.Air
       if airborne then
           JumpCircleAlpha = math.min(1, JumpCircleAlpha + 0.12)
       else
           JumpCircleAlpha = math.max(0, JumpCircleAlpha - 0.08)
       end
       local rp = RaycastParams.new()
       rp.FilterDescendantsInstances = {char}
       rp.FilterType = Enum.RaycastFilterType.Exclude
       local res = Workspace:Raycast(hrp.Position, Vector3.new(0, -20, 0), rp)
       local groundY = res and res.Position.Y or (hrp.Position.Y - 3)
       local col = GetJumpCircleColor()
       local radius = math.max(1, JumpCircleEff.Radius)
       local n = #JumpCircleParts
       local alpha = JumpCircleAlpha
       local basePos = Vector3.new(hrp.Position.X, groundY + 0.1, hrp.Position.Z)
       for i, p in ipairs(JumpCircleParts) do
           local ang = (i / n) * math.pi * 2
           local offset = Vector3.new(math.cos(ang) * radius, 0, math.sin(ang) * radius)
           local tangent = Vector3.new(-math.sin(ang), 0, math.cos(ang))
           local pos = basePos + offset
           p.CFrame = CFrame.lookAt(pos, pos + tangent)
           p.Color = col
           p.Transparency = 1 - alpha * 0.8
           local segLen = (2 * math.pi * radius / n) * 0.85
           p.Size = Vector3.new(0.18, 0.08, segLen)
       end
   end)
end)

function SaveConfig(name)
   if not name or name == "" then return false end
   if not writefile then return false end
   local data = {}
   for key, tbl in pairs(ConfigTables) do
       local copy = {}
       for k, v in pairs(tbl) do
           if type(v) == "number" or type(v) == "string" or type(v) == "boolean" then
               copy[k] = v
           end
       end
       data[key] = copy
   end
   local ok = pcall(function()
       writefile(CFG_FOLDER .. "/" .. name .. ".json", HttpService:JSONEncode(data))
   end)
   return ok
end

function LoadConfig(name)
   if not name or name == "" then return false end
   if not readfile or not isfile then return false end
   local path = CFG_FOLDER .. "/" .. name .. ".json"
   if not isfile(path) then return false end
   local ok, data = pcall(function()
       return HttpService:JSONDecode(readfile(path))
   end)
   if not ok or type(data) ~= "table" then return false end
   for key, saved in pairs(data) do
       local tbl = ConfigTables[key]
       if tbl and type(saved) == "table" then
           for k, v in pairs(saved) do
               tbl[k] = v
           end
       end
   end
   pcall(RefreshAllUI)
   pcall(GunModsTick)
   return true
end

function DeleteConfig(name)
   if not name or name == "" then return false end
   if not delfile or not isfile then return false end
   local path = CFG_FOLDER .. "/" .. name .. ".json"
   if not isfile(path) then return false end
   pcall(function() delfile(path) end)
   return true
end

function ListConfigs()
   local out = {}
   if not listfiles then return out end
   pcall(function()
       for _, f in ipairs(listfiles(CFG_FOLDER)) do
           local n = f:match("([^/\\]+)%.json$")
           if n then table.insert(out, n) end
       end
   end)
   table.sort(out)
   return out
end
